  // [666] FORM DE PROMOCIÓN — 3 caminos: sugerencia aceptada · manual guiado
  //   por estrategia (paso cero) · manual libre. Con vigencia por fechas + horario,
  //   panel de margen en vivo y guarda la jugada usada (mos.promociones.estrategia).
  // ════════════════════════════════════════════════════════════════
  function promoPintaEstrategias() {
    const g = $('promoEstrGrid');
    if (!g) return;
    const cur = _promoState.estrategia || '';
    g.innerHTML = PROMO_JUGADAS.filter(j => !j.guia).map(j => `
      <button type="button" class="promo-estr ${cur === j.k ? 'on' : ''}" onclick="MOS.promoSetEstrategia('${j.k}')">
        <span class="pe-ico">${j.ico}</span>
        <span class="pe-nom">${j.nom}</span>
        <span class="pe-cua">${j.cuando}</span>
      </button>`).join('')
      + `<button type="button" class="promo-estr ${cur === '' ? 'on' : ''}" onclick="MOS.promoSetEstrategia('')">
        <span class="pe-ico">✏️</span><span class="pe-nom">Libre</span>
        <span class="pe-cua">Arma la promo desde cero, sin plantilla.</span></button>`;
    const chip = $('promoEstrChip');
    if (chip) {
      const j = _PJ(cur);
      chip.innerHTML = j ? `<span class="pc-chip" style="background:rgba(99,102,241,.14);color:#a5b4fc;border-color:rgba(99,102,241,.4)">${j.ico} ${j.nom}</span>
        <button type="button" class="pc-mini" style="margin-left:6px" onclick="MOS.promoVerJugadas()">cambiar</button>` : '';
    }
  }
  function promoVerJugadas() {
    const g = $('promoEstrGrid');
    if (g) g.classList.remove('hidden');
    promoPintaEstrategias();
  }

  // Elegir jugada = pre-configurar el form para esa jugada
  function promoSetEstrategia(k) {
    _promoState.estrategia = k || '';
    const j = _PJ(k);
    const g = $('promoEstrGrid');
    if (g && k !== null) g.classList.add('hidden');
    promoPintaEstrategias();
    if (j && j.tipo) {
      const radio = document.querySelector('input[name="promoTipo"][value="' + j.tipo + '"]');
      if (radio) { radio.checked = true; promoSetTipo(j.tipo); }
      if (j.tipo === 'GRUPO') {
        promoSetModo('TOTAL');
        const cm = $('promoCantMin'); if (cm) cm.value = j.cant || 2;
      } else if (j.cant) {
        const cm = $('promoCantMin'); if (cm) cm.value = j.cant;
      }
    }
    if (j && j.hora) promoPresetHora('custom', j.hora[0], j.hora[1]);
    // sugerencia de descuento según el margen del producto ya elegido
    promoSugerirValor();
    promoRecalcMargen();
    promoActualizarEjemplo();
  }

  function _promoProdSel() {
    const sku = ($('promoSkuBase') || {}).value || '';
    if (!sku) return null;
    return (S.productos || []).find(pr => (pr.skuBase || pr.idProducto) === sku) || null;
  }
  // .90 / .50 hacia abajo (mismo criterio que mos._promo_psico del SQL 665)
  function _promoPsico(x) {
    const v = +x || 0;
    if (v <= 0.60) return 0.50;
    const f = Math.floor(v), d = v - f;
    return Math.max(0.50, f + (d >= 0.70 ? 0.90 : (d >= 0.40 ? 0.50 : -0.10)));
  }

  // Descuento sugerido según margen real del producto y la jugada elegida
  function promoSugerirValor() {
    const p = _promoProdSel();
    if (!p) return;
    const tipo = document.querySelector('input[name="promoTipo"]:checked')?.value;
    const pv = parseFloat(p.precioVenta) || 0, pc = parseFloat(p.precioCosto) || 0;
    if (pv <= 0) return;
    const mg = (pc > 0) ? ((pv - pc) / pv) * 100 : null;
    const k = _promoState.estrategia || '';
    let pct;
    if (k === 'escalera' || k === 'remate') pct = mg != null ? Math.min(45, Math.max(20, Math.floor(mg))) : 20;
    else if (k === 'sustitutos')            pct = 15;
    else                                    pct = mg != null ? Math.min(30, Math.max(10, Math.floor(mg) - 12)) : 10;
    const val = $('promoValor');
    if (!val || (val.value && parseFloat(val.value) > 0)) return;   // no pisar lo que el dueño ya escribió
    if (tipo === 'PORCENTAJE') val.value = pct;
    else if (tipo === 'GRUPO') {
      const n = parseFloat(($('promoCantMin') || {}).value) || 2;
      const total = _promoPsico(pv * (n - 1) + pv * (1 - pct / 100));
      val.value = _money(total).toFixed(2);
    }
  }

  // Panel de margen en vivo: precio normal → precio promo → margen % y en S/
  function promoRecalcMargen() {
    const box = $('promoMargenBox');
    if (!box) return;
    const tipo = document.querySelector('input[name="promoTipo"]:checked')?.value;
    const p = _promoProdSel();
    const pintar = (html) => { box.innerHTML = html; };
    if (tipo === 'COMBO') {
      const total = parseFloat(($('promoComboPrecio') || {}).value) || 0;
      let sumaPv = 0, sumaPc = 0, faltaCosto = false;
      (_promoState.comboItems || []).forEach(it => {
        const pr = (S.productos || []).find(x => (x.skuBase || x.idProducto) === it.skuBase);
        const c = parseFloat(it.cantidad) || 1;
        if (!pr) { faltaCosto = true; return; }
        sumaPv += (parseFloat(pr.precioVenta) || 0) * c;
        const pc = parseFloat(pr.precioCosto) || 0;
        if (pc > 0) sumaPc += pc * c; else faltaCosto = true;
      });
      if (!total || !sumaPv) { pintar(''); return; }
      const mg = (!faltaCosto && sumaPc > 0) ? ((total - sumaPc) / total) * 100 : null;
      pintar(_promoMargenHtml(sumaPv, total, mg, (!faltaCosto && sumaPc > 0) ? _money(total - sumaPc) : null, total < sumaPc && !faltaCosto, 'suelto', 'combo'));
      return;
    }
    if (!p) { pintar(''); return; }
    const pv = parseFloat(p.precioVenta) || 0, pc = parseFloat(p.precioCosto) || 0;
    const n  = Math.max(1, parseFloat(($('promoCantMin') || {}).value) || 1);
    const val = parseFloat(($('promoValor') || {}).value) || 0;
    if (!pv || !val) { pintar(''); return; }
    let totalPromo, totalNormal, unidades;
    if (tipo === 'PORCENTAJE') { unidades = 1; totalNormal = pv; totalPromo = pv * (1 - val / 100); }
    else {
      unidades = n;
      totalNormal = pv * n;
      const esTotal = $('promoModoTotal')?.classList.contains('active');
      totalPromo = esTotal ? val : val * n;
    }
    const costoTot = pc > 0 ? pc * unidades : 0;
    const mg = costoTot > 0 ? ((totalPromo - costoTot) / totalPromo) * 100 : null;
    pintar(_promoMargenHtml(totalNormal, totalPromo, mg, costoTot > 0 ? _money(totalPromo - costoTot) : null,
      costoTot > 0 && totalPromo < costoTot, unidades > 1 ? ('x' + unidades) : 'unidad', tipo === 'PORCENTAJE' ? 'unidad' : 'pack'));
  }

  function _promoMargenHtml(normal, promo, mgPct, mgSoles, perdida, escala, modo) {
    const dto = normal > 0 ? Math.max(0, (1 - promo / normal) * 100) : 0;
    const col = perdida ? '#f87171' : (mgPct == null ? '#94a3b8' : (mgPct >= 20 ? '#34d399' : (mgPct >= 10 ? '#fbbf24' : '#fb923c')));
    const txt = perdida ? '⚠ VENDE A PÉRDIDA'
      : (mgPct == null ? 'margen no verificable (producto sin costo cargado)'
        : `margen ${mgPct.toFixed(1)}% · S/ ${_money(mgSoles).toFixed(2)}`);
    return `<div class="promo-mg ${perdida ? 'bad' : ''}">
      <div style="display:flex;align-items:center;gap:10px;flex-wrap:wrap">
        <div><div class="pc-precio-lbl">normal ${escala}</div><div class="pc-precio-old">S/ ${_money(normal).toFixed(2)}</div></div>
        <div style="color:#475569">→</div>
        <div><div class="pc-precio-lbl">${modo === 'combo' ? 'combo' : (modo === 'pack' ? 'pack' : 'promo')}</div>
             <div class="pc-precio-new">S/ ${_money(promo).toFixed(2)}</div></div>
        <div style="flex:1;text-align:right">
          <span class="pc-chip" style="background:rgba(167,139,250,.14);color:#c4b5fd;border-color:rgba(167,139,250,.4)">−${dto.toFixed(0)}%</span>
          <div style="margin-top:4px"><span class="pc-chip" style="background:${col}1f;color:${col};border-color:${col}55">${txt}</span></div>
        </div>
      </div>
      ${perdida ? '<div style="font-size:10.5px;color:#fca5a5;margin-top:6px">Estás vendiendo por debajo de tu costo. Si es un remate a propósito, tendrás que confirmarlo dos veces.</div>' : ''}
    </div>`;
  }

  // ── Vigencia: presets ────────────────────────────────────────────
  function promoPresetVig(dias) {
    const d = new Date();
    const desde = _hoyISO();
    const h = new Date(d.getTime() + (dias * 86400000));
    const hasta = h.getFullYear() + '-' + String(h.getMonth() + 1).padStart(2, '0') + '-' + String(h.getDate()).padStart(2, '0');
    if ($('promoFDesde')) $('promoFDesde').value = desde;
    if ($('promoFHasta')) $('promoFHasta').value = hasta;
    document.querySelectorAll('#promoVigPresets .promo-modo-btn').forEach(b => b.classList.toggle('active', +b.dataset.d === dias));
  }

  // ── Horario: presets (null = todo el día) ────────────────────────
  function promoPresetHora(k, hd, hh) {
    const dd = $('promoHDesde'), hhEl = $('promoHHasta'), box = $('promoHoraCustom');
    const set = (a, b) => { if (dd) dd.value = a || ''; if (hhEl) hhEl.value = b || ''; };
    if (k === 'todo')        set('', '');
    else if (k === 'manana') set('06:00', '12:00');
    else if (k === 'tarde')  set('14:00', '18:00');
    else                     set(hd || (dd && dd.value) || '14:00', hh || (hhEl && hhEl.value) || '18:00');
    if (box) box.classList.toggle('hidden', k === 'todo');
    document.querySelectorAll('#promoHoraPresets .promo-modo-btn').forEach(b => b.classList.toggle('active', b.dataset.h === k));
    _promoState.horaPreset = k;
  }

  // ── Abrir el form (3 caminos) ────────────────────────────────────
  function promoAbrirNueva(estrategia, sug) {
    promoNuevoForm();
    _promoState.sugRef = sug || null;
    _promoState.estrategia = estrategia || '';
    const g = $('promoEstrGrid');
    if (g) g.classList.toggle('hidden', !!estrategia || !!sug);
    promoPintaEstrategias();
    if (estrategia) promoSetEstrategia(estrategia);
    if (sug) promoPrefill(sug);
    else promoRecalcMargen();
  }

  // Precargar el form desde una sugerencia del radar (el dueño ajusta y confirma)
  function promoPrefill(s) {
    const radio = document.querySelector('input[name="promoTipo"][value="' + s.tipo + '"]');
    if (radio) { radio.checked = true; promoSetTipo(s.tipo); }
    if (s.tipo === 'COMBO') {
      _promoState.comboItems = (s.items || []).map(i => ({ skuBase: i.skuBase, descripcion: i.descripcion, cantidad: i.cantidad || 1 }));
      _renderPromoComboItems();
      if ($('promoComboPrecio')) $('promoComboPrecio').value = _money(s.precioSugerido).toFixed(2);
    } else {
      if ($('promoSkuBase')) $('promoSkuBase').value = s.skuBase || '';
      if ($('promoSeleccionadoNombre')) $('promoSeleccionadoNombre').textContent = `${s.descripcion} (${s.skuBase})`;
      $('promoSeleccionado')?.classList.remove('hidden');
      if ($('promoBuscar')) $('promoBuscar').value = s.descripcion || '';
      if ($('promoCantMin')) $('promoCantMin').value = s.cantMin || 1;
      if (s.tipo === 'GRUPO') { promoSetModo('TOTAL'); if ($('promoValor')) $('promoValor').value = _money(s.precioSugerido).toFixed(2); }
      else { promoSetModo('UNITARIO'); if ($('promoValor')) $('promoValor').value = s.valorPromo; }
    }
    if ($('promoDesc')) $('promoDesc').value = s.descripcionSugerida || '';
    promoPresetVig(s.regla === 'REMATE' ? 14 : 30);
    if (s.horaDesde && s.horaHasta) promoPresetHora('custom', s.horaDesde, s.horaHasta);
    else promoPresetHora('todo');
    const nota = $('promoSugNota');
    if (nota) {
      nota.classList.remove('hidden');
      nota.innerHTML = `<div style="font-size:10.5px;color:#a5b4fc;font-weight:700;margin-bottom:3px">${s.emoji} ${s.etiqueta} · por qué esta jugada</div>
        <div style="font-size:11px;color:#94a3b8;line-height:1.5">${_escapeHtml(s.porqueDetalle || s.porque || '')}</div>`;
    }
    promoActualizarEjemplo();
    promoRecalcMargen();
  }
