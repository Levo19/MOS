  // [695] rotación semanal (misma fuente que el chip del card: series de S.rotacion)
  function _promoRotSem(p) {
    try {
      const prods = (S.rotacion && S.rotacion.productos) || {};
      const serie = prods[String(p.codigoBarra || '').trim()] || [];
      if (!serie.length) return 0;
      let totU = 0, totKg = 0;
      serie.forEach(x => { totU += parseFloat(x.unidades) || 0; totKg += parseFloat(x.kg) || 0; });
      const esPeso = !!_UMS_PESO[String(p.unidad || p.Unidad_Medida || '').trim().toUpperCase()];
      return (esPeso && totKg > 0 ? totKg : totU) / (serie.length || 8);
    } catch (_) { return 0; }
  }

  function _promoEstado(p) {
    const hoy = _hoyISO();
    const hasta = String(p.vigenciaHasta || '').substring(0, 10);
    if (hasta && hasta < hoy) return { k: 'vencida', txt: '⌛ vencida', col: '#94a3b8' };

  function _promoNombre(p) {
    if (p.tipo === 'COMBO') {
      const its = (p.items || []).map(i => i.descripcion).filter(Boolean);
      return its.length ? its.join(' + ') : `Combo de ${(p.items || []).length} productos`;
    }
    const prod = (S.productos || []).find(pr => (pr.skuBase || pr.idProducto) === p.skuBase);
    return prod ? (prod.descripcion || p.skuBase) : (p.skuBase || '—');
  }
  function _promoResumen(p) {
    const v = parseFloat(p.valorPromo) || 0;
    if (p.tipo === 'COMBO')  return `🛒 Combo a S/ ${_money(v).toFixed(2)}`;
    if (p.tipo === 'GRUPO')  return `📦 Lleva ${p.cantMin} por S/ ${_money((parseFloat(p.cantMin) || 0) * v).toFixed(2)} (c/u S/ ${_money(v).toFixed(2)})`;
    return `% −${v}% desde ${p.cantMin} ${(parseFloat(p.cantMin) || 0) === 1 ? 'unidad' : 'unidades'}`;
  }
  function _promoVentana(p) {
    const d = String(p.vigenciaDesde || '').substring(0, 10), h = String(p.vigenciaHasta || '').substring(0, 10);
    const fechas = (d || h) ? `📅 ${d || '—'} → ${h || 'sin fin'}` : '📅 sin vigencia definida';

    return fechas + horas;
  }

  async function _pcCargarPromos() {
    try {
      const r = await API.get('getPromociones', {});
      _promoState.lista = Array.isArray(r) ? r : ((r && r.data) || []);
      _promoState.lastFetch = Date.now();
      _promoSaveCache(_promoState.lista);
    } catch (e) { _promoState.lastError = e && e.message ? e.message : String(e); }
    if ($('pcBody')) _pcPinta();
  }

  async function _pcCargarSugerencias(seed) {
    _pcState.cargando = true; _pcState.err = '';
    if ($('pcBody') && _pcState.tab === 'sug') _pcPinta();
    try {
      const p = { n: 6 };
      if (seed) p.seed = seed;
      const r = await API.get('getPromoSugerencias', p);
      // el contrato _fetch('GET') devuelve r.data → acá necesitamos también el playbook,
      // así que la RPC lo manda dentro de data cuando viene por wrapper. Soportamos ambos.
      if (Array.isArray(r)) { _pcState.sug = r; }
      else if (r && Array.isArray(r.data)) { _pcState.sug = r.data; _pcState.playbook = r.playbook || _pcState.playbook; _pcState.seed = r.seed || ''; }
      else { _pcState.sug = []; }
    } catch (e) {
      _pcState.err = e && e.message ? e.message : String(e);
      _pcState.sug = [];
    }
    _pcState.cargando = false;
    if ($('pcBody')) _pcPinta();
  }

            <button class="pc-mini ${j.guia ? '' : 'pc-mini-ok'}" style="flex:none" onclick="MOS.pcUsarJugada('${j.k}')">${j.guia ? '🛡 Regla' : 'Usar esta jugada'}</button>
          </div>
        </div>`).join('');
  }

  function pcUsarJugada(k) {
    const j = _PJ(k);
    if (!j) return;
    if (j.guia) { toast('🛡 Es una regla de defensa: el top-seller nunca entra a promo. El radar ya lo excluye solo.', 'ok'); return; }
    $('promoCentro')?.remove();
    promoAbrirNueva(k);
  }

  function promoDesdeSugerencia(id) {
    const s = (_pcState.sug || []).find(x => x.id === id);
    if (!s) return;
    $('promoCentro')?.remove();
    promoAbrirNueva(s.estrategia, s);
  }

  async function promoDescartarSug(id) {
    const s = (_pcState.sug || []).find(x => x.id === id);
    if (!s) return;
    _pcState.sug = _pcState.sug.filter(x => x.id !== id);
    _pcPinta();
    toast('Idea descartada — no te la vuelvo a proponer por 30 días', 'ok');
    try { await API.post('promoDescartar', { skuBase: s.skuBase, regla: s.regla, por: 'MOS' }); }
    catch (e) { toast('No pude guardar el descarte: ' + e.message, 'error'); }
  }

  // ↻ Reactivar: abre el form con los datos de la promo pasada + vigencia nueva
  function promoReactivar(idRef) {
    const p = (_promoState.lista || []).find(x => (x.idPromo === idRef) || (x.skuBase === idRef));
    if (!p) return;
    $('promoCentro')?.remove();
    promoEditar(idRef);
    promoPresetVig(30);
    const t = $('promoTogActiva'); if (t) t.classList.add('on');
    toast('Ajusta la vigencia y confirma para volver a activarla', 'ok');
  }

  async function promoEliminarDirecto(idRef) {
    const p = (_promoState.lista || []).find(x => (x.idPromo === idRef) || (x.skuBase === idRef));
    if (!p) return;
    if (!await _modalConfirm('Se quita del POS de inmediato.', { danger: true, titulo: '¿Eliminar esta promoción?', okText: 'Eliminar' })) return;
    const backup = (_promoState.lista || []).slice();
    _promoState.lista = backup.filter(x => x !== p);
    _promoSaveCache(_promoState.lista); _pcPinta();
    toast('Promoción eliminada', 'ok');
    try { await API.post('eliminarPromocion', { idPromo: p.idPromo, skuBase: p.skuBase }); _pcCargarPromos(); }
    catch (e) { _promoState.lista = backup; _promoSaveCache(_promoState.lista); _pcPinta(); toast('Error: ' + e.message, 'error'); }
  }
